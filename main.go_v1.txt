package main

import (
	"fmt"
	"log"
	"strings"
	"time"

	"github.com/gocolly/colly/v2"
)

// Book representa a estrutura dos dados que vamos extrair
type Book struct {
	Title        string
	Price        string // Na etapa 2 vamos limpar e converter isso para float
	Rating       int
	Availability string
	ImageURL     string
}

// mapRating converte a classe CSS em texto para um número inteiro
func mapRating(classStr string) int {
	ratings := map[string]int{
		"One":   1,
		"Two":   2,
		"Three": 3,
		"Four":  4,
		"Five":  5,
	}
	
	// A classe vem como "star-rating Three", vamos pegar a segunda palavra
	parts := strings.Split(classStr, " ")
	if len(parts) == 2 {
		return ratings[parts[1]]
	}
	return 0
}

func main() {
	// 1. Inicializa o Colly
	c := colly.NewCollector(
		colly.AllowedDomains("books.toscrape.com"),
		// DICA DE OURO: User-Agent identificável (Requisito 4 do PDF)
		colly.UserAgent("Trainee-Scraper-Bot/1.0 (+https://meu-portfolio.com)"),
	)

	// DICA DE OURO: Rate Limiting para não bombardear o servidor
	c.Limit(&colly.LimitRule{
		DomainGlob:  "*books.toscrape.com*",
		Delay:       1 * time.Second, // 1 segundo de espera entre requisições
		RandomDelay: 500 * time.Millisecond,
	})

	// Slice para guardar nossos livros (usaremos na etapa 2)
	var books []Book

	// 2. Lógica de Extração: Encontra cada livro na listagem
	c.OnHTML("article.product_pod", func(e *colly.HTMLElement) {
		
		// Extraindo os dados usando seletores CSS baseados no HTML do PDF
		title := e.ChildAttr("h3 a", "title")
		price := e.ChildText(".price_color")
		
		// Limpando texto de disponibilidade (ex: "   In stock   " -> "In stock")
		availability := strings.TrimSpace(e.ChildText(".instock.availability"))
		
		// Imagem e Rating
		imageURL := e.ChildAttr(".image_container img", "src")
		ratingClass := e.ChildAttr("p.star-rating", "class")

		// Montando o objeto
		book := Book{
			Title:        title,
			Price:        price,
			Rating:       mapRating(ratingClass),
			Availability: availability,
			// AbsoluteURL resolve links relativos (ex: ../../img.jpg vira http://...)
			ImageURL:     e.Request.AbsoluteURL(imageURL),
		}

		books = append(books, book)
		
		// Imprime no console (Requisito da Etapa 1)
		fmt.Printf("Extraído: %s | Preço: %s | Estrelas: %d\n", book.Title, book.Price, book.Rating)
	})

	// 3. Lógica de Paginação (Prevenção de Loop Infinito)
	c.OnHTML("li.next a", func(e *colly.HTMLElement) {
		nextPage := e.Attr("href")
		// Converte o link relativo para absoluto antes de visitar
		absoluteURL := e.Request.AbsoluteURL(nextPage)
		
		fmt.Printf("\n---> Indo para a próxima página: %s\n\n", absoluteURL)
		
		// Visita a próxima página
		err := c.Visit(absoluteURL)
		if err != nil {
			log.Println("Erro ao visitar próxima página:", err)
		}
	})

	// Callbacks para logar quando a requisição começa e evitar que achemos que travou
	c.OnRequest(func(r *colly.Request) {
		fmt.Println("Visitando:", r.URL.String())
	})

	// 4. Start (Inicia o Crawler pela página 1)
	startURL := "https://books.toscrape.com/catalogue/page-1.html"
	err := c.Visit(startURL)
	if err != nil {
		log.Fatal("Erro fatal ao iniciar:", err)
	}

	fmt.Printf("\n=== RESUMO ===\nTotal de livros extraídos: %d\n", len(books))
}
